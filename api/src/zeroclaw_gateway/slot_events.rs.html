1//! Slot-scoped event payload helpers for the dashboard multiplex.
2//!
3//! M2 introduces a second class of event on `AppState.event_tx`: events that
4//! carry a `slot_id` so a dashboard client subscribing in read-only mode can
5//! filter by slot. Legacy `/ws/chat` events (no `slot_id`) flow unchanged.
6//!
7//! Helpers here produce the JSON shapes the dashboard WebSocket protocol
8//! expects (see `multi-session-dashboard.md §4.3`). Each helper returns a
9//! `serde_json::Value` ready to publish onto the broadcast channel.
10
11use serde_json::{Value, json};
12
13use crate::slot::SlotResponse;
14
15/// Event channel name used by the subscribe-mode filter.
16///
17/// The dashboard subscribes to symbolic channel names; the filter converts
18/// those to event-type prefixes (`"slots"`, `"slot"`, `"chat:<id>"`, etc.).
19pub const CHANNEL_SLOTS: &str = "slots";
20pub const CHANNEL_DASHBOARD: &str = "dashboard";
21
22/// Derived channel name for per-slot chat stream subscriptions.
23///
24/// Dashboard clients subscribe to `chat:<slot_id>` to receive chat deltas
25/// for a specific slot. Returns the channel identifier string.
26pub fn chat_channel(slot_id: &str) -> String {
27    format!("chat:{slot_id}")
28}
29
30/// Build a `slots` full-list event payload.
31///
32/// This is the payload shape for a full sidebar hydration event. M2's
33/// `ws::handle_dashboard_subscribe` intentionally does NOT push this on
34/// connect — clients hydrate via `GET /api/slots` and then receive
35/// incremental `slot_updated` / `chat_delta` events over the broadcast
36/// bus. M2.5 may promote this to an auto-emission on subscribe once the
37/// warm-agent refactor lands and full-list events are cheap to build on
38/// the hot path.
39pub fn slots_full(slots: &[SlotResponse]) -> Value {
40    json!({
41        "type": "slots",
42        "data": slots,
43    })
44}
45
46/// Build a single-slot update event payload.
47///
48/// Emitted whenever a slot transitions state (Idle → Running, Running →
49/// Idle, etc.) or its metadata changes. Carries `slot_id` at the event
50/// root to keep filter logic trivial.
51pub fn slot_updated(slot: &SlotResponse) -> Value {
52    json!({
53        "type": "slot",
54        "slot_id": slot.id,
55        "data": slot,
56    })
57}
58
59/// Build a chat-delta event payload for a slot's streaming turn.
60///
61/// `role` is one of `"user"`, `"assistant"`, `"tool"` (mirrors the existing
62/// WS chat protocol at `ws.rs`). `content` is the delta text; `done`
63/// signals the terminal event.
64pub fn chat_delta(slot_id: &str, role: &str, content: &str, done: bool) -> Value {
65    json!({
66        "type": "chat",
67        "slot_id": slot_id,
68        "data": {
69            "role": role,
70            "content": content,
71            "done": done,
72        },
73    })
74}
75
76/// Build a tool-approval request event tagged with the originating slot.
77///
78/// Companion to the connection-scoped `WsApprovalChannel` from #6387: when
79/// a slot-spawned agent fires `TurnEvent::ApprovalRequest`, the slot agent
80/// wraps it with `slot_id` before publishing so the dashboard sidebar can
81/// badge the correct slot.
82pub fn permission_request(
83    slot_id: &str,
84    request_id: &str,
85    tool_name: &str,
86    arguments_summary: &str,
87    timeout_secs: u64,
88) -> Value {
89    json!({
90        "type": "permission_request",
91        "slot_id": slot_id,
92        "data": {
93            "request_id": request_id,
94            "tool_name": tool_name,
95            "arguments_summary": arguments_summary,
96            "timeout_secs": timeout_secs,
97        },
98    })
99}
100
101/// Inspect an event payload and return the channel name a dashboard
102/// subscriber would use to receive it, if any.
103///
104/// Returns `None` for legacy /ws/chat events (no `slot_id`, no recognized
105/// dashboard `type`), which callers treat as "not intended for dashboard
106/// subscribers".
107pub fn event_channel(event: &Value) -> Option<String> {
108    let ty = event.get("type").and_then(|v| v.as_str())?;
109    match ty {
110        "slots" => Some(CHANNEL_SLOTS.to_string()),
111        "slot" => Some(CHANNEL_SLOTS.to_string()),
112        "dashboard" => Some(CHANNEL_DASHBOARD.to_string()),
113        "chat" | "permission_request" => {
114            let slot_id = event.get("slot_id").and_then(|v| v.as_str())?;
115            Some(chat_channel(slot_id))
116        }
117        _ => None,
118    }
119}

121#[cfg(test)]
122mod tests {
123    use super::*;
124    use crate::slot::SlotResponse;
125    use zeroclaw_infra::slot::{Slot, SlotState};
126
127    fn sample_slot_response() -> SlotResponse {
128        SlotResponse::from(Slot {
129            id: "slot-1".into(),
130            session_id: "gw_abc".into(),
131            title: "T".into(),
132            agent_config: Default::default(),
133            state: SlotState::Running,
134            created_at: 0,
135            updated_at: 0,
136            message_count: 0,
137            dirty: false,
138            workspace: None,
139        })
140    }
141
142    #[test]
143    fn slot_updated_carries_slot_id_at_root() {
144        let ev = slot_updated(&sample_slot_response());
145        assert_eq!(ev["type"], "slot");
146        assert_eq!(ev["slot_id"], "slot-1");
147        assert_eq!(ev["data"]["id"], "slot-1");
148        assert_eq!(ev["data"]["state"], "running");
149    }
150
151    #[test]
152    fn chat_delta_shape_matches_ws_protocol() {
153        let ev = chat_delta("slot-1", "assistant", "hello", false);
154        assert_eq!(ev["type"], "chat");
155        assert_eq!(ev["slot_id"], "slot-1");
156        assert_eq!(ev["data"]["role"], "assistant");
157        assert_eq!(ev["data"]["content"], "hello");
158        assert_eq!(ev["data"]["done"], false);
159    }
160
161    #[test]
162    fn permission_request_includes_request_id() {
163        let ev = permission_request("slot-1", "req-abc", "shell", "ls -la", 120);
164        assert_eq!(ev["type"], "permission_request");
165        assert_eq!(ev["slot_id"], "slot-1");
166        assert_eq!(ev["data"]["request_id"], "req-abc");
167        assert_eq!(ev["data"]["tool_name"], "shell");
168    }
169
170    #[test]
171    fn event_channel_routes_slot_events_to_slots_channel() {
172        let ev = slot_updated(&sample_slot_response());
173        assert_eq!(event_channel(&ev).as_deref(), Some("slots"));
174    }
175
176    #[test]
177    fn event_channel_routes_chat_to_per_slot_channel() {
178        let ev = chat_delta("slot-42", "assistant", "hi", true);
179        assert_eq!(event_channel(&ev).as_deref(), Some("chat:slot-42"));
180    }
181
182    #[test]
183    fn event_channel_routes_permission_request_to_per_slot_channel() {
184        let ev = permission_request("slot-42", "r", "t", "a", 30);
185        assert_eq!(event_channel(&ev).as_deref(), Some("chat:slot-42"));
186    }
187
188    #[test]
189    fn event_channel_returns_none_for_legacy_events() {
190        let legacy = json!({"type": "turn_start", "session_id": "gw_xyz"});
191        assert_eq!(event_channel(&legacy), None);
192    }
193
194    #[test]
195    fn slots_full_wraps_list() {
196        let s = sample_slot_response();
197        let ev = slots_full(&[s]);
198        assert_eq!(ev["type"], "slots");
199        let arr = ev["data"].as_array().unwrap();
200        assert_eq!(arr.len(), 1);
201        assert_eq!(arr[0]["id"], "slot-1");
202    }
