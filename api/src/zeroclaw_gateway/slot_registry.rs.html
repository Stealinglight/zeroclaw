1//! Warm-agent registry keyed on slot id (M2.5).
2//!
3//! Each slot owns a long-lived `Arc<Mutex<Agent>>` so a sequence of
4//! turns against the same slot reuses one Agent — skipping the
5//! expensive `Agent::from_config` init each time (see
6//! `multi-session-dashboard.md §4.5`). The shared `Arc<McpRegistry>`
7//! that all warm agents reference keeps the process's MCP subprocess
8//! count constant at N (configured servers), not N × slot_count.
9//!
10//! Scope for this M2.5 slice:
11//!   - `get_or_spawn(slot_id, overrides, base_config, shared_mcp)` —
12//!     returns a cloned `Arc<SlotEntry>`; constructs on miss, reuses on
13//!     hit.
14//!   - `get(slot_id)` — non-destructive lookup (used by REST
15//!     `/approve` to access the slot's pending-approval map).
16//!   - `remove(slot_id)` — forced eviction from `DELETE /api/slots/{id}`.
17//!   - `evict_idle()` — background sweep drops entries past `idle_ttl`.
18//!   - `pending_approvals` per slot: the slot-scoped map that the
19//!     per-turn `WsApprovalChannel` + REST `/approve` share.
20//!
21//! Pressure-eviction (LRU of idle entries when the store hits its hard
22//! limit) is a plan §4.5 concern but deferred here: the M1 store-level
23//! hard limit on create already prevents unbounded slot counts, so
24//! warm-agent pressure-eviction is a subsidiary safety net that lands
25//! with real load-testing work.
26

39/// A warm entry in the registry. `agent` is `Arc<tokio::sync::Mutex<_>>`
40/// so callers hold it across `.await` points on the single turn they're
41/// serving without blocking the registry itself.
42pub struct SlotEntry {
43    pub agent: Arc<tokio::sync::Mutex<Agent>>,
44    /// Last time this slot was active; idle sweep reads it via `touch`.
45    /// Wrapped in `Mutex` because entries live behind `Arc<SlotEntry>` —
46    /// interior mutability lets every turn-start cache hit and approval
47    /// resolution re-stamp it without needing `&mut`. A future
48    /// `AtomicU64`-of-unix-millis variant (deferred per plan §4.5) would
49    /// drop the lock; for now the contention surface is tiny (one store
50    /// per turn, one read per 60s sweep).
51    pub last_active: Mutex<Instant>,
52    /// Config frozen at spawn time. Edits to `config.toml` after this
53    /// do not retroactively reshape running slots (documented contract;
54    /// plan §4.5).
55    pub config_snapshot: SlotAgentConfig,
56    /// Slot-scoped pending-approval map. Shared between the per-turn
57    /// `WsApprovalChannel` (inserts) and `POST /api/slots/{id}/approve`
58    /// (pops by `request_id`).
59    pub pending_approvals: PendingApprovals,
60}
61
62impl SlotEntry {
63    /// Mark this slot as active now. Called on every turn start (via
64    /// `get_or_spawn` cache hit) and on `/approve` resolution so
65    /// `evict_idle` measures "time since last activity" rather than
66    /// "time since spawn" — without this, a slot parked on a long tool
67    /// approval would be silently evicted while the user was still
68    /// deciding.
69    pub fn touch(&self) {
70        *self
71            .last_active
72            .lock()
73            .expect("slot_entry last_active lock poisoned") = Instant::now();
74    }
75}

77/// Process-global registry of warm slot agents. Stored on
78/// `AppState::slot_registry`; cheap to clone (inner is one `Arc`).
79#[derive(Clone)]
80pub struct SlotRegistry {
81    inner: Arc<Inner>,
82}
83
84struct Inner {
85    slots: Mutex<HashMap<String, Arc<SlotEntry>>>,
86    idle_ttl: Duration,
87}
88
89impl SlotRegistry {
90    /// Build an empty registry. `idle_ttl_secs` mirrors the session
91    /// queue's default (600s) so sidebar-idle slots evict at the same
92    /// cadence the serialization slots clean up.
93    pub fn new(idle_ttl_secs: u64) -> Self {
94        Self {
95            inner: Arc::new(Inner {
96                slots: Mutex::new(HashMap::new()),
97                idle_ttl: Duration::from_secs(idle_ttl_secs),
98            }),
99        }
100    }
101
102    /// Get or spawn the warm agent for `slot_id`, applying the slot's
103    /// `SlotAgentConfig` overrides to `base_config` on first spawn.
104    ///
105    /// Spawn path calls `Agent::from_config_with_shared_mcp_backchannel`
106    /// so the agent reuses the caller's shared MCP subprocess tree and
107    /// surfaces tool approvals via the dashboard's operator-present
108    /// back-channel.
109    pub async fn get_or_spawn(
110        &self,
111        slot_id: &str,
112        overrides: &SlotAgentConfig,
113        base_config: Config,
114        shared_mcp: Option<Arc<McpRegistry>>,
115    ) -> anyhow::Result<Arc<SlotEntry>> {
116        if let Some(entry) = self.get(slot_id) {
117            // Cache-hit path: mark the slot as freshly active so the
118            // idle sweep doesn't evict a slot that's actively in use.
119            // Critical for turns that park on tool approvals longer than
120            // `idle_ttl` — the next reach-in here (e.g. an /approve
121            // followed by a new /messages) resets the clock.
122            entry.touch();
123            return Ok(entry);
124        }
125
126        // Build the agent outside the lock so two distinct slots can
127        // spawn concurrently. Racy double-spawn for the same slot: the
128        // loser's Agent drops on scope exit; we prefer the extant
129        // entry.
130        //
131        // Personality wiring (M4a): the slot's
132        // `agent_config.personality` filename — when set — restricts
133        // the agent's prompt-time identity load to that one allowlisted
134        // file. Frozen at spawn per the §4.5 freeze-at-spawn contract;
135        // editing the slot's personality after the agent is warm has no
136        // effect until the slot is evicted and re-spawned.
137        let effective_config = apply_slot_overrides(base_config, overrides);
138        let agent = Agent::from_config_with_shared_mcp_backchannel_personality(
139            &effective_config,
140            None,
141            shared_mcp,
142            overrides.personality.clone(),
143        )
144        .await?;
145        let agent = Arc::new(tokio::sync::Mutex::new(agent));
146
147        let entry = Arc::new(SlotEntry {
148            agent,
149            last_active: Mutex::new(Instant::now()),
150            config_snapshot: overrides.clone(),
151            pending_approvals: new_pending_approvals(),
152        });
153
154        {
155            let mut map = self
156                .inner
157                .slots
158                .lock()
159                .expect("slot_registry lock poisoned");
160            if let Some(existing) = map.get(slot_id) {
161                return Ok(existing.clone());
162            }
163            map.insert(slot_id.to_string(), entry.clone());
164        }
165
166        Ok(entry)
167    }
168
169    /// Non-destructive lookup. Returns a cloned `Arc<SlotEntry>` when
170    /// the slot has a warm agent, `None` otherwise.
171    ///
172    /// Used by `POST /api/slots/{id}/approve` to read slot-scoped
173    /// `pending_approvals` without triggering a spawn.
174    pub fn get(&self, slot_id: &str) -> Option<Arc<SlotEntry>> {
175        let map = self
176            .inner
177            .slots
178            .lock()
179            .expect("slot_registry lock poisoned");
180        map.get(slot_id).cloned()
181    }
182
183    /// Drop entries that haven't been touched within `idle_ttl`.
184    ///
185    /// Returns the count of evictions. Wired to a background tokio task
186    /// ticking every 60s at gateway startup.
187    pub fn evict_idle(&self) -> usize {
188        let now = Instant::now();
189        let ttl = self.inner.idle_ttl;
190        let mut map = self
191            .inner
192            .slots
193            .lock()
194            .expect("slot_registry lock poisoned");
195        let before = map.len();
196        map.retain(|_, entry| {
197            let last = *entry
198                .last_active
199                .lock()
200                .expect("slot_entry last_active lock poisoned");
201            now.duration_since(last) <= ttl
202        });
203        before - map.len()
204    }
205
206    /// Forcibly drop a slot's warm agent. Called when the slot is
207    /// deleted from the store so we don't hold an Agent referring to
208    /// config that no longer corresponds to a user-visible slot.
209    pub fn remove(&self, slot_id: &str) -> Option<Arc<SlotEntry>> {
210        let mut map = self
211            .inner
212            .slots
213            .lock()
214            .expect("slot_registry lock poisoned");
215        map.remove(slot_id)
216    }
217
218    /// Count of currently warm slots. Exposed for tests and future
219    /// metrics.
220    pub fn len(&self) -> usize {
221        self.inner
222            .slots
223            .lock()
224            .expect("slot_registry lock poisoned")
225            .len()
226    }
227
228    /// True when no slots are warm.
229    pub fn is_empty(&self) -> bool {
230        self.len() == 0
231    }
232}
233
234impl Default for SlotRegistry {
235    fn default() -> Self {
236        Self::new(600)
237    }
238}

275    /// Constructing a real `Agent` in a unit test pulls a full provider
276    /// stack; instead we exercise `Mutex<Instant>` semantics directly.
277    /// The behaviour under test is "store through `&self` and read the
278    /// new value back" — identical semantics to `SlotEntry::touch`,
279    /// minus the Agent. This guards the lock-expect contract and the
280    /// interior-mutability shape `touch` relies on.
281    #[test]
282    fn mutex_instant_allows_touch_through_shared_ref() {
283        let stamp = std::sync::Mutex::new(Instant::now());
284        let before = *stamp.lock().unwrap();
285        // Busy-sleep long enough that `Instant` advances monotonically.
286        std::thread::sleep(Duration::from_millis(5));
287        // Simulate SlotEntry::touch — store through a shared ref.
288        *stamp.lock().unwrap() = Instant::now();
289        let after = *stamp.lock().unwrap();
290        assert!(
291            after > before,
292            "touch must produce a strictly later Instant"
293        );
294    }
295
296    /// Asserts the `retain` closure in `evict_idle` respects a touched
297    /// timestamp — entries whose `last_active` is within `ttl` of `now`
298    /// must be kept. This is a direct mirror of the predicate at
299    /// `evict_idle`'s `map.retain` call, and the key regression this PR
300    /// fix prevents: before the fix, `last_active` was set at spawn and
301    /// never updated, so an active slot would still be evicted once
302    /// `ttl` elapsed from spawn time regardless of touches.
303    #[test]
304    fn evict_predicate_retains_recently_touched_entry() {
305        let now = Instant::now();
306        let ttl = Duration::from_secs(60);
307
308        // Entry touched within TTL: must be retained.
309        let fresh_last_active = now - Duration::from_secs(5);
310        assert!(now.duration_since(fresh_last_active) <= ttl);
311
312        // Entry untouched for longer than TTL: must be evicted.
313        // (Use a timestamp from before `now` to avoid negative-duration
314        // panics on monotonic clocks.)
315        let stale_last_active = now - Duration::from_secs(120);
316        assert!(now.duration_since(stale_last_active) > ttl);
317    }
