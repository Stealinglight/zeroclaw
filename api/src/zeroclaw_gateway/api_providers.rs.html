1//! `GET /api/providers` — list configured model providers.
2//!
3//! The dashboard's slot settings drawer needs the list of providers a
4//! user has actually configured (via `[providers.models.*]` in
5//! `config.toml` or the `/api/config/*` surface) so the Advanced tab
6//! can render a real dropdown rather than hardcoding the provider zoo.
7//! ZeroClaw ships ~15 first-class provider implementations; treating
8//! the configured map as the source of truth keeps that list out of
9//! the frontend.
10//!
11//! Response shape is intentionally minimal — `id`, `display_name`,
12//! `model`, `is_fallback`. Auth headers and bespoke per-provider
13//! options (api keys, base URLs) stay out of this surface; those are
14//! the `/api/config/*` shape's concern.
15

29/// One configured provider entry.
30#[derive(Debug, Clone, Serialize, Deserialize)]
31#[cfg_attr(feature = "schema-export", derive(JsonSchema))]
32pub struct ProviderInfo {
33    /// Map key from `[providers.models.<id>]` — e.g. `"anthropic"`,
34    /// `"claude_code"`, `"openai_codex"`. The slot's
35    /// `agent_config.provider` references this id verbatim.
36    pub id: String,
37    /// Human-readable label for the dropdown. Falls back to the id
38    /// when the static lookup has no entry.
39    pub display_name: String,
40    /// The currently configured model for this provider entry, if any.
41    /// `None` means the provider was registered without a default
42    /// model — the slot's `agent_config.model` must fill the gap.
43    #[serde(skip_serializing_if = "Option::is_none")]
44    pub model: Option<String>,
45    /// `true` when this provider is the gateway's fallback (i.e.
46    /// `providers.fallback == Some(self.id)`). Frontend uses this to
47    /// label the default option in the dropdown.
48    pub is_fallback: bool,
49}
50
51/// `GET /api/providers` response.
52#[derive(Debug, Clone, Serialize, Deserialize, Default)]
53#[cfg_attr(feature = "schema-export", derive(JsonSchema))]
54pub struct ProviderListResponse {
55    pub providers: Vec<ProviderInfo>,
56}

58/// Map a provider id to a human-readable label. Returns the id itself
59/// when no mapping exists — keeps unconfigured-yet-discovered ids
60/// rendering legibly in the UI.
61fn display_name_for(id: &str) -> &'static str {
62    match id {
63        "anthropic" => "Anthropic",
64        "azure_openai" => "Azure OpenAI",
65        "bedrock" => "AWS Bedrock",
66        "claude_code" => "Claude Code",
67        "compatible" => "OpenAI-Compatible",
68        "copilot" => "GitHub Copilot",
69        "gemini" => "Google Gemini",
70        "gemini_cli" => "Gemini CLI",
71        "glm" => "GLM",
72        "kilocli" => "Kilo CLI",
73        "llamacpp" => "llama.cpp",
74        "ollama" => "Ollama",
75        "openai" => "OpenAI",
76        "openai_codex" => "OpenAI Codex",
77        "openrouter" => "OpenRouter",
78        "telnyx" => "Telnyx",
79        _ => "",
80    }
81}

83/// `GET /api/providers` — list every entry in `[providers.models]`.
84///
85/// Empty list when no providers are configured (which is also when
86/// the gateway can't actually answer model requests — UI can use the
87/// empty state to nudge the user toward `/onboarding`).
88pub async fn handle_api_providers_list(
89    State(state): State<AppState>,
90    headers: HeaderMap,
91) -> Response {
92    if let Err(e) = require_auth(&state, &headers) {
93        return e.into_response();
94    }
95
96    let cfg = state.config.lock();
97    let fallback = cfg.providers.fallback.clone();
98
99    // Sort by id for stable ordering across requests — the map is a
100    // HashMap and iteration order would otherwise jitter, which makes
101    // diff-based dropdowns ugly and tests flaky.
102    let mut entries: Vec<(String, &zeroclaw_config::schema::ModelProviderConfig)> = cfg
103        .providers
104        .models
105        .iter()
106        .map(|(k, v)| (k.clone(), v))
107        .collect();
108    entries.sort_by(|a, b| a.0.cmp(&b.0));
109
110    let providers = entries
111        .into_iter()
112        .map(|(id, entry)| {
113            let label = display_name_for(&id);
114            let display_name = if label.is_empty() {
115                id.clone()
116            } else {
117                label.to_string()
118            };
119            ProviderInfo {
120                is_fallback: fallback.as_deref() == Some(id.as_str()),
121                model: entry.model.clone(),
122                display_name,
123                id,
124            }
125        })
126        .collect();
127
128    Json(ProviderListResponse { providers }).into_response()
129}

256    async fn body_to_json(response: Response) -> serde_json::Value {
257        let bytes = to_bytes(response.into_body(), usize::MAX)
258            .await
259            .unwrap_or_default();
260        serde_json::from_slice(&bytes).unwrap_or(serde_json::Value::Null)
261    }

265        let state = provider_test_state(Config::default());

273    #[tokio::test]
274    async fn providers_returns_configured_entries_sorted() {
275        let mut cfg = Config::default();
276        cfg.providers.models.insert(
277            "openai".into(),
278            ModelProviderConfig {
279                model: Some("gpt-5".into()),
280                ..Default::default()
281            },
282        );
283        cfg.providers.models.insert(
284            "anthropic".into(),
285            ModelProviderConfig {
286                model: Some("claude-sonnet-4".into()),
287                ..Default::default()
288            },
289        );
290        cfg.providers.fallback = Some("anthropic".into());
291
292        let state = provider_test_state(cfg);
293        let resp = handle_api_providers_list(State(state), HeaderMap::new()).await;
294        assert_eq!(resp.status(), axum::http::StatusCode::OK);
295        let json = body_to_json(resp).await;
296        let providers = json["providers"].as_array().expect("providers array");
297        assert_eq!(providers.len(), 2);
298        // Sort order is alphabetical by id.
299        assert_eq!(providers[0]["id"], "anthropic");
300        assert_eq!(providers[0]["display_name"], "Anthropic");
301        assert_eq!(providers[0]["model"], "claude-sonnet-4");
302        assert_eq!(providers[0]["is_fallback"], true);
303        assert_eq!(providers[1]["id"], "openai");
304        assert_eq!(providers[1]["display_name"], "OpenAI");
305        assert_eq!(providers[1]["model"], "gpt-5");
306        assert_eq!(providers[1]["is_fallback"], false);
307    }
